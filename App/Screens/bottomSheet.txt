import React, { useRef } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import BottomSheet from 'react-native-simple-bottom-sheet';

const App = () => {
  const panelRef = useRef(null);

  return (
    <View style={styles.container}>
      <View style={styles.content}>
        <Text style={styles.title}>Welcome to Bottom Sheet Example</Text>
      </View>
      <TouchableOpacity
        style={styles.button}
        onPress={() => {
          if (panelRef.current) {
            panelRef.current.togglePanel();
          }
        }}
      >
        <Text style={styles.buttonText}>Toggle Bottom Sheet</Text>
      </TouchableOpacity>
      <BottomSheet
        ref={(ref) => (panelRef.current = ref)}
        isOpen={false} // Initially closed
        sliderMinHeight={0} // Hidden when closed
        sliderMaxHeight={400} // Maximum height when open
      >
        <View style={styles.bottomSheetContent}>
          <Text style={styles.sheetText}>
            Key Features of the Code:
            - Button to Toggle Bottom Sheet: The TouchableOpacity triggers the togglePanel method on the BottomSheet reference.
            - Initial State Management: The bottom sheet is closed (isOpen=false) when the app starts.
            - Custom Styles: Styled components for a user-friendly interface.
            - Dynamic Height: The bottom sheet has a sliderMinHeight and sliderMaxHeight to allow for expandable content.
          </Text>
        </View>
      </BottomSheet>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
    backgroundColor: '#f5f5f5',
  },
  content: {
    marginBottom: 20,
  },
  title: {
    fontSize: 18,
    fontWeight: 'bold',
    textAlign: 'center',
  },
  button: {
    backgroundColor: '#6200ee',
    paddingVertical: 10,
    paddingHorizontal: 20,
    borderRadius: 8,
  },
  buttonText: {
    color: '#fff',
    fontSize: 16,
    fontWeight: 'bold',
  },
  bottomSheetContent: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
  },
  sheetText: {
    fontSize: 16,
    color: '#333',
  },
});

export default App;
