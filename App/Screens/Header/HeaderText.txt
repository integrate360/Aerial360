import React, { useContext, useState } from 'react';
import {
  Text,
  View,
  Image,
  Dimensions,
  Modal,
  TouchableWithoutFeedback,
  Animated,
  StyleSheet,
  ScrollView,
  Alert,
  TouchableOpacity,
} from 'react-native';
import {
  Appbar,
  IconButton,
  Badge,
  Drawer,
  Divider,
  Icon,
} from 'react-native-paper';
import { useNavigation } from '@react-navigation/native';
import UserContext from '../../context/UserContext';
import { Colors } from '../Config/Config';
import fonts from '../FontFamily/FontFamily';

const { width, height } = Dimensions.get('window');
const isTablet = width > 600;

const HeaderComponent = ({ DataFunc, onOpenCalendar, TrueFun, Arrow, hide }) => {
  const [notifications, setNotifications] = useState(5);
  const [modalVisible, setModalVisible] = useState(false);
  const translateX = useState(new Animated.Value(-width * 0.8))[0];
  const navigation = useNavigation();
  const { logout } = useContext(UserContext);

  const openDrawer = () => {
    setModalVisible(true);
    Animated.timing(translateX, {
      toValue: 0,
      duration: 300,
      useNativeDriver: true,
    }).start();
  };

  const closeDrawer = () => {
    Animated.timing(translateX, {
      toValue: -width * 0.8,
      duration: 300,
      useNativeDriver: true,
    }).start(() => {
      setModalVisible(false);
    });
  };

  const LogOut = () => {
    Alert.alert(
      'Confirm Logout', // Title of the alert
      'Are you sure you want to log out?', // Message in the alert
      [
        {
          text: 'Cancel',
          onPress: () => console.log('Logout cancelled'), // Optional action for cancel
          style: 'cancel', // Styling for cancel button
        },
        {
          text: 'OK',
          onPress: () => {
            // setModalVisible(false); // Close the modal
            // logout(); // Call the logout function
            setModalVisible(false);
            logout();
            navigation.navigate('Login');
          },
        },
      ],
      { cancelable: false }, // Prevent closing the alert by tapping outside
    );
  };

  return (
    <>
      <Appbar.Header
        style={{
          backgroundColor: '#fff',
          shadowColor: '#874d3b',
          shadowOffset: { width: 0, height: 10 },
          shadowOpacity: 0.6,
          shadowRadius: 50,
          elevation: 10,
          height: height * 0.08,
        }}>
        <View style={{ flexDirection: 'row', alignItems: 'center', flex: 1 }}>
          {Arrow == true ? (
            <Appbar.BackAction onPress={() => navigation.goBack()} />
          ) : (
            <IconButton
              icon="menu"
              color="#000"
              size={isTablet ? 40 : 30}
              onPress={openDrawer}
            />
          )}

          <Image
            source={require('../../assets/AirbLogo.png')}
            style={{
              width: isTablet ? width * 0.15 : width * 0.25,
              height: isTablet ? height * 0.06 : height * 0.05,
              resizeMode: 'contain',
              marginLeft: width * 0.02,
            }}
          />
        </View>

        {TrueFun === true ? (
          <IconButton
            icon="calendar"
            size={28}
            color="black"
            onPress={() => onOpenCalendar()}
          />
        ) : (
          <View
            style={{
              flexDirection: 'row',
              alignItems: 'center',
              position: 'absolute',
              right: width * 0.03,
              top: height * 0.01,
            }}>
            {hide == true ? null : (
              <TouchableOpacity
                onPress={() => navigation.navigate('SearchClasses')}>
                <Image
                  source={require('../../assets/search.png')}
                  width={20}
                  style={{ width: 21, height: 21, top: 10, right: 10 }}
                />
              </TouchableOpacity>
            )}
          </View>
        )}
      </Appbar.Header>

      {/* Modal for Side Drawer */}
      <Modal
        transparent
        visible={modalVisible}
        animationType="none"
        onRequestClose={closeDrawer}>
        <TouchableWithoutFeedback onPress={closeDrawer}>
          <View style={styles.overlay} />
        </TouchableWithoutFeedback>

        {/* Drawer Content */}
        <Animated.View style={[styles.drawer, { transform: [{ translateX }] }]}>
          <View style={{ flex: 1, backgroundColor: '#fff' }}>
            {/* Drawer Header */}
            <View style={styles.drawerHeader}>
              <Image
                source={require('../../assets/Airbound-Logo-.png')}
                style={{
                  width: '100%',
                  height: '100%',
                }}
              />
            </View>

            <IconButton
              icon="close"
              size={30}
              style={styles.drawerCloseButton}
              iconColor="#fff"
              onPress={closeDrawer}
            />

            {/* Drawer Items */}

            <ScrollView
              contentContainerStyle={styles.scrollViewContent}
              showsVerticalScrollIndicator={false}>
              <Drawer.Section style={styles.drawerSection} showDivider={false}>
                {[
                  {
                    label: 'How It Works',
                    icon: 'table',
                    screen: 'How_it_Works',
                  },

                  {
                    label: 'Our Trainers',
                    icon: 'account-group',
                    screen: 'Our_Trainers',
                  },
                  {
                    label: 'Bookings',
                    icon: 'calendar-check',
                    screen: 'Booking',
                  },
                  {
                    label: 'Gifting',
                    icon: 'package',
                    screen: 'Gifting',
                  },
                  {
                    label: 'Terms & Conditions',
                    icon: 'forum',
                    screen: 'TERMSCompnent',
                  },
                  {
                    label: 'FAQ',
                    icon: 'forum',
                    screen: 'FAQsection',
                  },
                ].map(item => (
                  <Drawer.Item
                    key={item.label}
                    // label={item.label}
                    style={{ right: 20 }}
                    labelMaxFontSizeMultiplier={23}
                    icon={({ color, size }) => (
                      <>
                        <IconButton
                          icon={item.icon}
                          iconColor="#fff"
                          size={21}
                          style={{
                            alignContent: 'center',
                            alignSelf: 'center',
                            alignItems: 'center',
                            backgroundColor: Colors.secondary,
                            borderRadius: 10,
                            opacity: 0.7,
                          }}
                        />
                        <Text
                          style={{
                            color: 'black',
                            marginLeft: 10,
                            fontSize: 15,
                            fontWeight: '600',
                            opacity: 0.8,
                            fontFamily: fonts.regular
                          }}>
                          {item?.label}
                        </Text>
                        <View
                          style={{
                            opacity: 0.8,
                          }}>
                          <IconButton
                            name="chevron-right"
                            size={25}
                            color="red"
                            style={{}}
                          />
                        </View>
                      </>
                    )}
                    onPress={() =>
                      navigation.navigate(item?.screen) + setModalVisible(false)
                    }
                    contentStyle={styles.drawerItemContent}
                  // right={() => (
                  //   <IconButton
                  //     icon={'chevron-right'}
                  //     iconColor="black"
                  //     size={21}
                  //     style={{alignSelf: 'flex-end', marginLeft: 50}}
                  //   />
                  // )}
                  />
                ))}
              </Drawer.Section>
            </ScrollView>

            {/* Footer */}
            <View style={{ flex: 1, justifyContent: 'flex-end' }}>
              <View
                style={{
                  borderWidth: 0.2,
                  borderColor: Colors.secondary,
                  opacity: 0.6,
                }}></View>
              <Drawer.Section
                style={styles.drawerFooterSection}
                showDivider={false}>
                <TouchableOpacity
                  style={{
                    flexDirection: 'row',
                    alignItems: 'center',
                    paddingVertical: 12,
                    paddingHorizontal: 16,
                  }}
                  onPress={() => logout()}
                >
                  <IconButton
        icon="exit-to-app" // React Native Paper icon name
        size={24}
        color={Colors.primary}
        style={{ marginRight: 3 }} // Adjust spacing if needed
      />
                  <Text
                    style={{
                      fontSize: 16,
                      color: 'black',
                      fontFamily: fonts.regular,
                    }}
                  >
                    Sign Out
                  </Text>
                </TouchableOpacity>
              </Drawer.Section>
            </View>
          </View>
        </Animated.View>
      </Modal>
    </>
  );
};

const styles = StyleSheet.create({
  overlay: {
    flex: 1,
    backgroundColor: 'rgba(0, 0, 0, 0.5)',
  },
  drawer: {
    position: 'absolute',
    left: 0,
    width: width * 0.8,
    height: '100%',
    backgroundColor: '#fff',
    shadowColor: '#000',
    shadowOffset: { width: 5, height: 0 },
    shadowOpacity: 0.2,
    shadowRadius: 10,
    elevation: 5,
  },
  drawerHeader: {
    backgroundColor: '#f2f2f2',
    height: height * 0.25,
    alignItems: 'center',
    justifyContent: 'center',
  },
  drawerCloseButton: {
    position: 'absolute',
    right: 10,
    top: 10,
    zIndex: 1,
  },
  scrollViewContent: {
    padding: 10,
  },
  drawerSection: {
    marginVertical: 10,
  },
  drawerItem: {
    marginVertical: 5,
    flexDirection: 'row',
    alignItems: 'center',
  },
  drawerFooterSection: {
    padding: 10, fontFamily: fonts.regular
  },
});

export default HeaderComponent;
