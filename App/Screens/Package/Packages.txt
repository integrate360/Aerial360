const {height} = Dimensions.get('window'); // Get screen height

const CustomBottomSheet = ({locations}) => {
  const translateY = useRef(new Animated.Value(height)).current; // Initially off-screen
  const [isOpen, setIsOpen] = useState(false);

  const snapPoints = [height * 0.25, height * 0.5, height * 0.9];

  const panResponder = useRef(
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onPanResponderMove: (event, gestureState) => {
        const newTranslateY = Math.max(0, translateY._value + gestureState.dy);
        translateY.setValue(newTranslateY);
      },
      onPanResponderRelease: (event, gestureState) => {
        const velocity = gestureState.vy;
        let finalPosition = snapPoints[0];

        if (velocity > 0.5) {
          finalPosition = snapPoints[2];
        } else if (velocity < -0.5) {
          finalPosition = snapPoints[0];
        } else {
          const nearestSnapPoint = snapPoints.reduce((prev, curr) =>
            Math.abs(curr - translateY._value) <
            Math.abs(prev - translateY._value)
              ? curr
              : prev,
          );
          finalPosition = nearestSnapPoint;
        }

        Animated.spring(translateY, {
          toValue: finalPosition,
          useNativeDriver: true,
        }).start(() => setIsOpen(finalPosition === snapPoints[0]));
      },
    }),
  ).current;

  const toggleSheet = () => {
    const toValue = isOpen ? height : snapPoints[0];
    Animated.spring(translateY, {
      toValue,
      useNativeDriver: true,
    }).start(() => setIsOpen(!isOpen));
  };

  const openGoogleMaps = link => {
    Linking.openURL(link).catch(err => console.error('An error occurred', err));
  };

  const renderLocation = ({item}) => (
    <View style={styles.card}>
      <Text style={styles.locationName}>{item.name}</Text>
      <Text style={styles.address}>{item.address}</Text>
      <TouchableOpacity
        style={styles.linkButton}
        onPress={() => openGoogleMaps(item.googleMapLink)}>
        <Text style={styles.linkText}>View on Google Maps</Text>
      </TouchableOpacity>
    </View>
  );

  return (
    <>
      <TouchableOpacity style={styles.toggleButton} onPress={toggleSheet}>
        <Text style={styles.toggleText}>
          {isOpen ? 'Close' : 'Open'} Locations
        </Text>
      </TouchableOpacity>
      <Animated.View
        style={[styles.bottomSheet, {transform: [{translateY}]}]}
        {...panResponder.panHandlers}>
        <View style={styles.contentContainer}>
          <Text style={styles.title}>Available Locations</Text>
          <FlatList
            data={locations}
            renderItem={renderLocation}
            keyExtractor={item => item._id}
            contentContainerStyle={styles.listContainer}
          />
        </View>
      </Animated.View>
    </>
  );
};
